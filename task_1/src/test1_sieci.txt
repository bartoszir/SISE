import numpy as np

""" funkcje aktywacji i ich pochodne """
def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def sigmoid_derivative(x):
    s = sigmoid(x)
    return s * (1 - s)

def tanh(x):
    return np.tanh(x)

def tanh_derivative(x):
    return 1 - np.tanh(x) ** 2

def relu(x):
    return np.maximum(0, x)

def relu_derivative(x):
    return (x > 0).astype(float)

    def forward(self, X):
        self.Z1 = np.dot(X, self.W1) + self.b1
        self.A1 = self.activation(self.Z1)
        self.Z2 = np.dot(self.A1, self.W2) + self.b2
        return self.Z2 # wyjscie liniowe

    def backward(self, X, Y, Y_pred):
        m = Y.shape[0]
        dZ2 = (Y_pred - Y) / m # MSE
        dW2 = self.A1.T.dot(dZ2)
        db2 = np.sum(dZ2, axis=0, keepdims=True)

        dA1 = np.dot(dZ2, self.W2.T)
        dZ1 = dA1 * self.activation_derivative(self.Z1)
        dW1 = np.dot(X.T, dZ1)
        db1 = np.sum(dZ1, axis=0, keepdims=True)

        # aktualizacja z momentum
        self.vW2 = self.momentum * self.vW2 - self.learning_rate * dW2
        self.vb2 = self.momentum * self.vb2 - self.learning_rate * db2
        self.vW1 = self.momentum * self.vW1 - self.learning_rate * dW1
        self.vb1 = self.momentum * self.vb1 - self.learning_rate * db1

        self.W2 += self.vW2
        self.b2 += self.vb2
        self.W1 += self.vW1
        self.b1 += self.vb2

    def train(self, X_train, Y_train, X_val, Y_val, epochs=100):
        train_errors = []
        val_errors = []

        for epoch in range(epochs):
            Y_pred = self.forward(X_train)
            self.backward(X_train, Y_train, Y_pred)

            train_mse = np.mean((Y_train - Y_pred) ** 2)
            val_mse = np.mean((Y_val - self.forward(X_val)) ** 2)

            train_errors.append(train_mse)
            val_errors.append(val_mse)

            if epoch % 10 == 0:
                print(f"Epoch {epoch}: Train MSE = {train_mse:.5f}, Test MSE = {val_mse:.5f}")

        return train_errors, val_errors

    def predict(self, X):
        return self.forward(X)
